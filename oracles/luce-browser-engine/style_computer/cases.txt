# The cascade oracle cases (r39): "css <text>" is the case's author style sheet ("\n" for a new
# line), "tree <markup>" div#root's content, and each "q <id> <property>..." line asks
# getComputedStyle(element) for the properties. The properties are ones whose computed value is
# the cascaded value as written (keywords, integers, numbers), its inherited value, or the
# initial value. "html", "body" and "root" name those elements.

# The user-agent style sheet alone.
css
tree <div id=a><span id=b></span><p id=c></p><em id=d></em><section id=e></section><ul id=f><li id=g></li></ul><b id=h></b><i id=i></i><cite id=j></cite></div>
q html display text-align visibility
q body display
q root display
q a display font-style
q b display font-style
q c display
q d display font-style
q e display
q f display list-style-position
q g display list-style-position
q h display
q i display font-style
q j display font-style

# Specificity: id > class > type; equal specificity, the later rule wins.
css div { display: inline } .k { display: inline-block } #x { display: none } p { visibility: hidden } p { visibility: collapse } span.k { cursor: pointer } .k.k { cursor: text } span { cursor: move }
tree <div id=x class=k></div><div id=y class=k></div><div id=z></div><p id=p></p><span id=s class=k></span><span id=t></span>
q x display
q y display
q z display
q p visibility
q s cursor
q t cursor

# !important beats specificity and order; the inline style beats the sheet; !important beats the inline style.
css #a { display: inline } div { display: block !important } .b { z-index: 3 !important } #b { z-index: 5 } #c { order: 2 } .d { order: 4 !important }
tree <div id=a></div><div id=b class=b style="z-index: 7"></div><div id=c style="order: 1"></div><div id=d class=d style="order: 9"></div><div id=e style="order: 6 !important" class=d></div>
q a display
q b z-index
q c order
q d order
q e order

# Inheritance: inherited properties take the parent's value, the others their initial value.
css #o { visibility: hidden; cursor: pointer; text-align: center; z-index: 4; text-transform: uppercase; font-style: italic; direction: rtl }
tree <div id=o><span id=p><b id=q></b></span></div>
q o visibility cursor text-align z-index text-transform font-style direction
q p visibility cursor text-align z-index text-transform font-style direction
q q visibility cursor text-align z-index text-transform font-style direction

# inherit, initial, unset.
css #o { z-index: 4; cursor: pointer; display: inline-block; opacity: 0.25 } #p { z-index: inherit; cursor: initial; display: inherit; opacity: inherit } #q { cursor: unset; z-index: unset; display: unset } #r { visibility: hidden } #s { visibility: initial } #t { visibility: unset }
tree <div id=o><span id=p><b id=q></b></span></div><div id=r><span id=s></span><span id=t></span></div>
q p z-index cursor display opacity
q q z-index cursor display opacity
q s visibility
q t visibility

# Cascade layers: unlayered normal declarations win; later layers win; for !important the order is reversed.
css @layer one, two;\n@layer two { #a { z-index: 2 } #b { z-index: 2 !important } }\n@layer one { #a { z-index: 1 } #b { z-index: 1 !important } #c { order: 1 } }\n#c { order: 5 } #a { z-index: 9 } #b { z-index: 9 !important } #d { order: 3 !important }\n@layer two { #d { order: 2 !important } }
tree <div id=a></div><div id=b></div><div id=c></div><div id=d></div>
q a z-index
q b z-index
q c order
q d order

# Nested and anonymous layers, and a layer statement fixing the order before the blocks.
css @layer outer { @layer inner { #a { order: 1 } #b { order: 1 } } #a { order: 2 } }\n@layer { #b { order: 3 } }\n@layer outer.inner { #c { order: 4 } }\n@layer { #c { order: 5 } }\n@layer late, early;\n@layer early { #d { z-index: 1 } }\n@layer late { #d { z-index: 2 } }
tree <div id=a></div><div id=b></div><div id=c></div><div id=d></div>
q a order
q b order
q c order
q d z-index

# Specificity inside a layer does not beat the layer order.
css @layer low { #a.x.y { cursor: pointer } }\n@layer high { div { cursor: text } }
tree <div id=a class="x y"></div>
q a cursor

# revert rolls back to the user-agent value; revert-layer to the previous layer.

tree <div id=a></div><div id=b></div><div id=c></div><div id=d></div><p id=e></p>
q a display
q b display
q c z-index
q d z-index
q e visibility

# Custom properties: cascade, inheritance and var() substitution.
css #o { --a: 10px; --b: var(--a); --c: foo  bar; --d: block } #p { --a: 20px; display: var(--d) } #q { --e: var(--missing, inline-block); display: var(--e) } #r { display: var(--nothing) } #s { z-index: var(--z, 8) }

q o --a --b --c --d
q p --a --b --c display
q q --a --e display
q r display
q s z-index

# Custom properties: !important, the inline style, and the last declaration winning.
css #a { --x: one !important } #a { --x: two } .b { --y: sheet } #c { --z: first; --z: second }
